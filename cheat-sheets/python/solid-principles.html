SOLID Principles — Python

Five principles for writing maintainable, extensible object-oriented Python code

S - Single Responsibility O - Open/Closed L - Liskov Substitution I - Interface Segregation D - Dependency Inversion ← Index
S

Single Responsibility Principle

A class should have only one reason to change. Separate concerns into focused classes.

# BAD: OrderProcessor does too many things
class OrderProcessor:
    def process(self, order: dict):
        # 1. Validate
        if not order.get("customer_id"):
            raise ValueError("No customer")

        # 2. Save to DB
        import sqlite3
        conn = sqlite3.connect("orders.db")
        conn.execute("INSERT INTO orders ...")
        conn.commit()

        # 3. Send confirmation email
        import smtplib
        server = smtplib.SMTP("smtp.gmail.com")
        server.sendmail(...)

        # 4. Log everything
        with open("orders.log", "a") as f:
            f.write(f"Order processed: {order}")
# 4 reasons to change this class!
# GOOD: Each class has one job
class OrderValidator:
    def validate(self, order: dict) -> None:
        if not order.get("customer_id"):
            raise ValueError("No customer")

class OrderRepository:
    def save(self, order: dict) -> int:
        # DB logic only
        return db.execute("INSERT ...")

class EmailService:
    def send_confirmation(self, order: dict) -> None:
        # Email logic only
        smtp.send(order["email"], "Confirmed!")

class OrderLogger:
    def log_order(self, order_id: int) -> None:
        logging.info(f"Order {order_id} processed")

class OrderProcessor:
    def __init__(self, validator, repository, email, logger):
        self.validator = validator
        self.repository = repository
        self.email = email
        self.logger = logger

    def process(self, order: dict) -> int:
        self.validator.validate(order)
        order_id = self.repository.save(order)
        self.email.send_confirmation(order)
        self.logger.log_order(order_id)
        return order_id

Python-specific tips

  • Functions in Python also follow SRP — a function should do one thing
  • Use dataclasses for pure data containers, separate from business logic
  • Modules are also a unit of responsibility; split large modules into packages
O

Open/Closed Principle

Open for extension, closed for modification. Add behavior by adding code, not changing it.

# BAD: Adding a new shape requires modifying this
def calculate_area(shapes: list) -> float:
    total = 0.0
    for shape in shapes:
        if shape["type"] == "circle":
            total += 3.14 * shape["radius"] ** 2
        elif shape["type"] == "rectangle":
            total += shape["width"] * shape["height"]
        elif shape["type"] == "triangle":
            total += shape["base"] * shape["height"] / 2
        # Need to add hexagon? Modify this function!
    return total
from abc import ABC, abstractmethod
from typing import Protocol
import math

# Using Protocol (Python 3.8+) - no inheritance needed
class Shape(Protocol):
    def area(self) -> float: ...

class Circle:
    def __init__(self, radius: float): self.radius = radius
    def area(self) -> float: return math.pi * self.radius ** 2

class Rectangle:
    def __init__(self, w: float, h: float): self.w, self.h = w, h
    def area(self) -> float: return self.w * self.h

# Adding hexagon? Just add a new class! No existing code changes.
class Hexagon:
    def __init__(self, side: float): self.side = side
    def area(self) -> float: return 3 * math.sqrt(3) / 2 * self.side ** 2

# Open for extension: new shapes work automatically
def calculate_total_area(shapes: list[Shape]) -> float:
    return sum(shape.area() for shape in shapes)

shapes = [Circle(5), Rectangle(3, 4), Hexagon(6)]
total = calculate_total_area(shapes)

Python OCP techniques

  • Use Protocols for structural subtyping — no forced inheritance
  • Use ABC for explicit contracts with abstract methods
  • Strategy pattern with callables: pass behavior as functions/lambdas
  • Plugin/registry pattern: dictionary mapping names to implementations
L

Liskov Substitution Principle

Subclasses must honor the contract established by the parent class.

# BAD: ReadOnlyFile violates File contract
class File:
    def read(self) -> str:
        return "content"

    def write(self, data: str) -> None:
        with open(self.path, "w") as f:
            f.write(data)

class ReadOnlyFile(File):
    def write(self, data: str) -> None:
        # Violates LSP: parent promises write works!
        raise PermissionError("File is read-only")

def process_file(f: File):
    content = f.read()
    f.write(content.upper())  # expects this to work!

# Breaks with ReadOnlyFile — LSP violated
process_file(ReadOnlyFile())
from abc import ABC, abstractmethod

# Better hierarchy: separate readable from writable
class ReadableFile(ABC):
    @abstractmethod
    def read(self) -> str: ...

class WritableFile(ReadableFile):
    @abstractmethod
    def write(self, data: str) -> None: ...

class RegularFile(WritableFile):
    def read(self) -> str:
        return open(self.path).read()
    def write(self, data: str) -> None:
        open(self.path, "w").write(data)

class ReadOnlyFile(ReadableFile):
    def read(self) -> str:
        return open(self.path).read()
    # No write() — it's not in the contract!

# Functions take only what they need
def read_content(f: ReadableFile) -> str:
    return f.read()  # works for both types

def update_content(f: WritableFile, data: str) -> None:
    f.write(data)  # only called with writable files

LSP in Python

  • Python's duck typing means LSP applies to protocols too — honor the interface
  • Don't raise exceptions for methods that the parent promises to work
  • Covariant return types and contravariant parameters are OK
  • mypy can help catch LSP violations with type annotations
I

Interface Segregation Principle

Clients should not be forced to depend on interfaces they don't use. Split fat interfaces.

from abc import ABC, abstractmethod

# BAD: Fat interface — not all printers can scan or fax!
class MultifunctionDevice(ABC):
    @abstractmethod
    def print_doc(self, document): ...

    @abstractmethod
    def scan_doc(self, document): ...

    @abstractmethod
    def fax_doc(self, document): ...

class OldPrinter(MultifunctionDevice):
    def print_doc(self, document): print("Printing...")

    def scan_doc(self, document):
        raise NotImplementedError("Can't scan!")  # Violation!

    def fax_doc(self, document):
        raise NotImplementedError("Can't fax!")   # Violation!
from typing import Protocol

# GOOD: Small, focused protocols
class Printer(Protocol):
    def print_doc(self, document) -> None: ...

class Scanner(Protocol):
    def scan_doc(self, document) -> None: ...

class FaxMachine(Protocol):
    def fax_doc(self, document) -> None: ...

# Simple printer only implements what it can do
class SimplePrinter:
    def print_doc(self, document) -> None:
        print(f"Printing: {document}")
    # No scan or fax — no empty stubs!

# MultiFunctionPrinter can do everything
class MultiFunctionPrinter:
    def print_doc(self, document) -> None: ...
    def scan_doc(self, document) -> None: ...
    def fax_doc(self, document) -> None: ...

# Functions ask for only what they need
def print_report(printer: Printer, doc) -> None:
    printer.print_doc(doc)

def scan_and_print(device: Printer, doc) -> None:
    # Works with any printer! Not just MultiFunctionPrinter
    device.print_doc(doc)

Python advantage: Protocols make ISP natural

  • Python's typing.Protocol enables structural subtyping — no forced inheritance
  • Duck typing already encourages small, focused interfaces
  • Abstract Base Classes from abc module for explicit contracts
  • Mix-ins can compose capabilities without fat base classes
D

Dependency Inversion Principle

High-level modules should not depend on low-level modules. Both should depend on abstractions.

# BAD: OrderService is tightly coupled to PostgreSQL
import psycopg2

class OrderService:
    def __init__(self):
        # Direct dependency on concrete implementation!
        self.conn = psycopg2.connect("postgresql://...")

    def get_order(self, order_id: int) -> dict:
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM orders WHERE id = %s", (order_id,))
        return cursor.fetchone()

    def create_order(self, data: dict) -> int:
        # Can't switch to MySQL, SQLite, or mock in tests!
        cursor = self.conn.cursor()
        cursor.execute("INSERT INTO orders ...")
        return cursor.lastrowid
from typing import Protocol

# Abstraction (the interface)
class OrderRepository(Protocol):
    def find_by_id(self, order_id: int) -> dict | None: ...
    def save(self, order: dict) -> int: ...

# Low-level: PostgreSQL implementation
class PostgresOrderRepository:
    def find_by_id(self, order_id: int) -> dict | None:
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM orders WHERE id = %s", (order_id,))
        return cursor.fetchone()

    def save(self, order: dict) -> int:
        cursor = self.conn.cursor()
        cursor.execute("INSERT INTO orders ...")
        return cursor.lastrowid

# In-memory for testing
class InMemoryOrderRepository:
    def __init__(self): self.orders: dict[int, dict] = {}
    def find_by_id(self, order_id: int) -> dict | None:
        return self.orders.get(order_id)
    def save(self, order: dict) -> int:
        order_id = len(self.orders) + 1
        self.orders[order_id] = order
        return order_id

# High-level: depends on abstraction
class OrderService:
    def __init__(self, repository: OrderRepository):  # injected!
        self.repository = repository

    def get_order(self, order_id: int) -> dict | None:
        return self.repository.find_by_id(order_id)

    def create_order(self, data: dict) -> int:
        return self.repository.save(data)

# Production
service = OrderService(PostgresOrderRepository())
# Tests
service = OrderService(InMemoryOrderRepository())

Dependency Injection in Python

  • Constructor injection (preferred): pass dependencies in __init__
  • Use default arguments for optional dependencies
  • DI frameworks: dependency_injector, injector, python-inject
  • pytest fixtures are essentially a DI system for tests
