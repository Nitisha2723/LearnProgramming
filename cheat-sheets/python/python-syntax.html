Function Definitions
# Basic
def greet(name: str, greeting: str = "Hello") -> str:
    return f"{greeting}, {name}!"

# *args: variable positional arguments
def sum_all(*args):
    return sum(args)

# **kwargs: variable keyword arguments
def configure(**kwargs):
    for key, value in kwargs.items():
        print(f"{key} = {value}")

# Keyword-only (after *)
def draw(x, y, *, color="red", size=5):
    pass
draw(1, 2, color="blue")  # must use keyword

# Positional-only (before /)
def add(x, y, /):
    return x + y
add(1, 2)  # must be positional

# Lambda
square = lambda x: x ** 2
sorted(lst, key=lambda x: x[1])

# Closures
def counter():
    count = 0
    def increment():
        nonlocal count
        count += 1
        return count
    return increment

# Generators
def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b
Decorators
from functools import wraps
import time

# Simple decorator
def timer(func):
    @wraps(func)  # preserves __name__, __doc__
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        print(f"{func.__name__} took {time.time()-start:.3f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)

# Decorator with arguments
def retry(times=3):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(times):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == times - 1: raise
            return None
        return wrapper
    return decorator

@retry(times=5)
def fetch_data(): ...

# Built-in decorators
@staticmethod   # no self/cls, utility methods
@classmethod    # cls instead of self, alternative constructors
@property       # getter/setter syntax
@functools.lru_cache(maxsize=128)  # memoization
@functools.cached_property          # computed once, then cached
Context Managers
# Built-in usage
with open("file.txt") as f:
    data = f.read()

# Custom context manager (class-based)
class DatabaseConnection:
    def __enter__(self):
        self.conn = connect_to_db()
        return self.conn

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.conn.close()
        return False  # don't suppress exceptions

with DatabaseConnection() as conn:
    conn.query("SELECT ...")

# Context manager using contextlib
from contextlib import contextmanager

@contextmanager
def timer():
    start = time.time()
    try:
        yield  # code inside with-block runs here
    finally:
        print(f"Elapsed: {time.time() - start:.3f}s")

with timer():
    expensive_operation()

# Multiple context managers
with open("in.txt") as f_in, open("out.txt", "w") as f_out:
    f_out.write(f_in.read())

Object-Oriented Programming

Classes & Inheritance
class Animal:
    # Class variable (shared)
    count = 0

    def __init__(self, name: str, age: int):
        self.name = name    # instance variable
        self._age = age     # "protected" (convention)
        Animal.count += 1

    @property
    def age(self) -> int:
        return self._age

    @age.setter
    def age(self, value: int):
        if value < 0: raise ValueError("Age can't be negative")
        self._age = value

    def speak(self) -> str:
        raise NotImplementedError

    def __repr__(self) -> str:
        return f"Animal(name={self.name!r}, age={self.age})"

    def __eq__(self, other) -> bool:
        return isinstance(other, Animal) and self.name == other.name

class Dog(Animal):
    def __init__(self, name, age, breed):
        super().__init__(name, age)
        self.breed = breed

    def speak(self) -> str:
        return f"{self.name} says Woof!"

    def __repr__(self):
        return f"Dog({self.name!r}, {self.breed!r})"

# Multiple inheritance
class ServiceDog(Dog, CertifiedWorker):
    def __init__(self, name, age, breed, certification):
        Dog.__init__(self, name, age, breed)
        self.certification = certification
Dunder / Magic Methods
class Vector:
    def __init__(self, x, y): self.x, self.y = x, y

    # String representations
    def __str__(self): return f"({self.x}, {self.y})"
    def __repr__(self): return f"Vector({self.x}, {self.y})"

    # Arithmetic
    def __add__(self, other): return Vector(self.x+other.x, self.y+other.y)
    def __sub__(self, other): return Vector(self.x-other.x, self.y-other.y)
    def __mul__(self, scalar): return Vector(self.x*scalar, self.y*scalar)
    def __rmul__(self, scalar): return self.__mul__(scalar)

    # Comparison
    def __eq__(self, other): return self.x == other.x and self.y == other.y
    def __lt__(self, other): return abs(self) < abs(other)

    # Container protocol
    def __len__(self): return 2
    def __getitem__(self, i): return (self.x, self.y)[i]

    # Numeric
    def __abs__(self): return (self.x**2 + self.y**2)**0.5
    def __bool__(self): return self.x != 0 or self.y != 0
    def __neg__(self): return Vector(-self.x, -self.y)

    # Context manager
    def __enter__(self): return self
    def __exit__(self, *args): pass

    # Callable
    def __call__(self): return self.__abs__()

    # Hashing (if __eq__ defined, must define __hash__)
    def __hash__(self): return hash((self.x, self.y))
ABCs & Protocols
from abc import ABC, abstractmethod
from typing import Protocol, runtime_checkable

# Abstract Base Class
class Shape(ABC):
    @abstractmethod
    def area(self) -> float: ...

    @abstractmethod
    def perimeter(self) -> float: ...

    def describe(self):  # concrete method
        print(f"Area: {self.area():.2f}")

class Circle(Shape):
    def __init__(self, r): self.r = r
    def area(self): return 3.14159 * self.r ** 2
    def perimeter(self): return 2 * 3.14159 * self.r

# Protocol (structural subtyping - duck typing)
@runtime_checkable
class Drawable(Protocol):
    def draw(self) -> None: ...
    def get_color(self) -> str: ...

class Button:  # doesn't explicitly inherit Drawable
    def draw(self): print("Drawing button")
    def get_color(self): return "blue"

isinstance(Button(), Drawable)  # True (runtime check!)

# __slots__ for memory efficiency
class Point:
    __slots__ = ['x', 'y']  # disables __dict__, saves memory
    def __init__(self, x, y):
        self.x, self.y = x, y

Exceptions

Exception Handling
try:
    result = 10 / 0
except ZeroDivisionError as e:
    print(f"Error: {e}")
except (TypeError, ValueError) as e:
    print(f"Type/Value error: {e}")
except Exception:
    raise  # re-raise
else:
    print("No exception!")  # runs if no exception
finally:
    print("Always runs")

# Raise exceptions
raise ValueError("Invalid input")
raise ValueError("Bad input") from original_error  # chaining

# Custom exception
class PaymentError(Exception):
    def __init__(self, message, code):
        super().__init__(message)
        self.code = code

# Exception groups (Python 3.11+)
try:
    raise ExceptionGroup("multiple", [ValueError("a"), TypeError("b")])
except* ValueError as eg:
    print("Handled ValueErrors:", eg.exceptions)

from typing import Optional, Union, Any, Callable

from typing import TypeVar, Generic
T = TypeVar('T')

class Stack(Generic[T]):
    def __init__(self) -> None: self.items: list[T] = []
    def push(self, item: T) -> None: self.items.append(item)
    def pop(self) -> T: return self.items.pop()
