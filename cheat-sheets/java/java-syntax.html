Primitive Types

Java's 8 primitive types with sizes

Local Variable Type InferenceJava 10+

Use var to let the compiler infer the type

var name = "Alice";           // String
var list = new ArrayList<String>();
var map = new HashMap<String, Integer>();
var entry = map.entrySet().iterator();

// Works in for loops

    System.out.println(item);
}

// Does NOT work for fields, parameters, or return types
var improves readability when the type is obvious from the right side

Control Flow

Conditionals
// Standard if-else
if (x > 0) {
    System.out.println("positive");
} else if (x == 0) {
    System.out.println("zero");
} else {
    System.out.println("negative");
}

// Ternary operator
String result = x > 0 ? "positive" : "non-positive";

// Switch expression (Java 14+)
String day = switch (dayNum) {
    case 1 -> "Monday";
    case 2 -> "Tuesday";
    case 3, 4, 5 -> "Midweek";
    default -> "Weekend";
};

// Switch with yield
int score = switch (grade) {
    case "A" -> 100;
    case "B" -> { int v = 80; yield v; }
    default -> 0;
};
Loops
// Traditional for
for (int i = 0; i < 10; i++) { ... }

// Enhanced for (for-each)
for (String s : list) { ... }
for (int x : array) { ... }

// While
while (condition) { ... }

// Do-while (executes at least once)
do {
    // code
} while (condition);

// Loop control
for (int i = 0; i < 100; i++) {
    if (i == 5) continue;  // skip this iteration
    if (i == 10) break;    // exit loop
}

// Labeled break (for nested loops)
outer:
for (int i = 0; i < 3; i++) {
    for (int j = 0; j < 3; j++) {
        if (i == 1 && j == 1) break outer;
    }
}
Pattern MatchingJava 16+
// Old way
if (obj instanceof String) {
    String s = (String) obj;
    System.out.println(s.length());
}

// Pattern matching (Java 16+)
if (obj instanceof String s) {
    System.out.println(s.length());
}

// In switch (Java 21+)
String result = switch (obj) {
    case Integer i -> "int: " + i;
    case String s -> "str: " + s;
    case null -> "null";
    default -> "other";
};
Pattern matching eliminates redundant casts

Object-Oriented Programming

Classes & Objects
public class Person {
    // Fields
    private final String name;  // immutable
    private int age;

    // Constructor
    public Person(String name, int age) {
        this.name = name;
        this.age = age;
    }

    // Getters
    public String getName() { return name; }
    public int getAge() { return age; }

    // Setter
    public void setAge(int age) {
        if (age < 0) throw new IllegalArgumentException();
        this.age = age;
    }

    @Override
    public String toString() {
        return "Person{name='" + name + "', age=" + age + "}";
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Person p)) return false;
        return age == p.age && Objects.equals(name, p.name);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, age);
    }
}
Interfaces & Abstract Classes
// Interface
public interface Drawable {
    void draw();  // abstract by default
    default String getType() { return "shape"; }  // default method
    static Drawable of(String type) { ... }       // static method
}

// Abstract class
public abstract class Shape {
    protected String color;
    public abstract double area();  // must override
    public void describe() {
        System.out.println("Area: " + area());
    }
}

// Implementing both
public class Circle extends Shape implements Drawable {
    private double radius;

    public Circle(double radius) { this.radius = radius; }

    @Override public double area() { return Math.PI * radius * radius; }
    @Override public void draw() { System.out.println("Drawing circle"); }
}
Prefer interfaces over abstract classes; use abstract classes for shared state
RecordsJava 16+
// Records: immutable data carriers
public record Point(int x, int y) {
    // Compact canonical constructor
    public Point {
        if (x < 0 || y < 0) throw new IllegalArgumentException();
    }

    // Custom methods
    public double distance() {
        return Math.sqrt(x * x + y * y);
    }
}

// Usage
var p = new Point(3, 4);
p.x();       // accessor (not getX)
p.y();       // accessor
p.equals(new Point(3, 4));  // true - by value
p.toString(); // "Point[x=3, y=4]"
Records auto-generate: constructor, accessors, equals, hashCode, toString
Sealed Classes/InterfacesJava 17+
// Sealed: restrict which classes can extend/implement
public sealed interface Shape
    permits Circle, Rectangle, Triangle {}

public record Circle(double radius) implements Shape {}
public record Rectangle(double w, double h) implements Shape {}
public final class Triangle implements Shape { ... }

// Works well with pattern matching
double area = switch (shape) {
    case Circle c -> Math.PI * c.radius() * c.radius();
    case Rectangle r -> r.w() * r.h();
    case Triangle t -> t.base() * t.height() / 2;
}; // no default needed - compiler knows all cases
Enums
public enum Planet {
    MERCURY(3.303e+23, 2.4397e6),
    VENUS  (4.869e+24, 6.0518e6),
    EARTH  (5.976e+24, 6.37814e6);

    private final double mass;
    private final double radius;

    Planet(double mass, double radius) {
        this.mass = mass;
        this.radius = radius;
    }

    static final double G = 6.67300E-11;
    double surfaceGravity() { return G * mass / (radius * radius); }
    double surfaceWeight(double otherMass) {
        return otherMass * surfaceGravity();
    }
}

// Usage
Planet.EARTH.surfaceWeight(75.0);
Planet.valueOf("EARTH");   // by name
Planet.values();           // all values
Planet.EARTH.name();       // "EARTH"
Planet.EARTH.ordinal();    // 2

Exception Handling

Try-Catch-Finally
// Basic try-catch
try {
    int result = 10 / 0;
} catch (ArithmeticException e) {
    System.err.println("Error: " + e.getMessage());
} finally {
    System.out.println("Always runs");
}

// Multi-catch (Java 7+)
try {
    // ...
} catch (IOException | SQLException e) {
    e.printStackTrace();
}

// Try-with-resources (Java 7+) - auto-closes
try (var in = new FileInputStream("file.txt");
     var out = new FileOutputStream("out.txt")) {
    // use in and out - closed automatically
} catch (IOException e) {
    e.printStackTrace();
}

// Custom exception
public class BusinessException extends RuntimeException {
    private final int code;
    public BusinessException(String message, int code) {
        super(message);
        this.code = code;
    }
    public int getCode() { return code; }
}

Functional Programming Java 8+

Lambda Expressions
// Lambda syntax
Runnable r = () -> System.out.println("run");
Comparator<String> cmp = (a, b) -> a.compareTo(b);
Predicate<Integer> isPos = n -> n > 0;
Function<String, Integer> len = String::length; // method ref

// Common functional interfaces
Predicate<T>     : T -> boolean
Function<T,R>    : T -> R
BiFunction<T,U,R>: (T,U) -> R
Consumer<T>      : T -> void
Supplier<T>      : () -> T
UnaryOperator<T> : T -> T
BinaryOperator<T>: (T,T) -> T

// Method references
String::length        // instance method
System.out::println   // instance method of specific obj
Integer::parseInt     // static method
ArrayList::new        // constructor ref
Optional
// Creating Optional
Optional<String> opt = Optional.of("hello");
Optional<String> empty = Optional.empty();
Optional<String> nullable = Optional.ofNullable(null);

// Consuming
opt.isPresent();                // true
opt.isEmpty();                  // false (Java 11+)
opt.get();                      // "hello" (throws if empty)
opt.orElse("default");          // "hello"
opt.orElseGet(() -> compute());
opt.orElseThrow(IllegalStateException::new);

// Transforming
opt.map(String::toUpperCase);   // Optional["HELLO"]
opt.filter(s -> s.length() > 3);
opt.flatMap(s -> findUser(s));

// Chaining
Optional.ofNullable(user)
        .map(User::getAddress)
        .map(Address::getCity)
        .orElse("Unknown");
Never use Optional.get() without checking isPresent()

CompletableFuture
// Async computation
CompletableFuture<String> future =
    CompletableFuture.supplyAsync(() -> fetchData());

// Chain operations
future
    .thenApply(data -> process(data))     // transform
    .thenAccept(result -> save(result))   // consume
    .thenRun(() -> notify())             // run after
    .exceptionally(e -> handleError(e)); // error handler

// Combine futures
CompletableFuture<String> f1 = fetchUser();
CompletableFuture<String> f2 = fetchOrders();

CompletableFuture.allOf(f1, f2)
    .thenRun(() -> combine(f1.join(), f2.join()));

CompletableFuture.anyOf(f1, f2)
    .thenAccept(result -> use(result));
